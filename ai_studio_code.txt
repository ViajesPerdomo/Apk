name: Construir APK de Viajes Perdomo

on:
  push:
    branches: [ "main", "master" ]
  pull_request:
    branches: [ "main", "master" ]
  workflow_dispatch:
    inputs:
      build_type:
        description: 'Tipo de compilación (debug o release)'
        required: true
        default: 'debug'
        type: choice
        options:
          - debug
          - release

jobs:
  build-apk:
    name: Compilar APK Android
    runs-on: ubuntu-latest

    steps:
      - name: 📥 Clonar repositorio
        uses: actions/checkout@v4

      - name: 📦 Extraer ZIP si el proyecto fue subido comprimido
        run: |
          if [ -f "project.zip" ]; then
            echo "Detectado project.zip, descomprimiendo..."
            unzip -o project.zip -d .
          elif ls *.zip 1> /dev/null 2>&1; then
            ZIP_FILE=$(ls *.zip | head -n 1)
            echo "Detectado archivo $ZIP_FILE, descomprimiendo..."
            unzip -o "$ZIP_FILE" -d .
          else
            echo "No se encontraron archivos zip en la raíz. Procediendo con el árbol git estándar."
          fi

      - name: ☕ Configurar Java 17 (Temurin - Sin conflictos)
        uses: actions/setup-java@v4
        with:
          distribution: 'temurin'
          java-version: '17'
          cache: 'gradle'

      - name: 🟢 Configurar Node.js 20 LTS
        uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: 'npm'

      - name: 📦 Instalar dependencias de npm
        run: |
          npm install --legacy-peer-deps

      - name: 🔨 Compilar aplicación web (Vite build)
        run: |
          npm run build

      - name: 📱 Inicializar y Sincronizar Capacitor Android
        run: |
          if [ ! -d "android" ]; then
            echo "Generando proyecto nativo Android..."
            npx cap add android
          fi
          echo "Sincronizando assets compilados con Android..."
          npx cap sync android

      - name: 🔧 Configurar permisos de ejecución en gradlew
        run: |
          if [ -f "android/gradlew" ]; then
            chmod +x android/gradlew
          else
            echo "Error: gradlew no encontrado"
            exit 1
          fi

      - name: 🚀 Compilar APK con Gradle
        run: |
          cd android
          ./gradlew assembleDebug --stacktrace --no-daemon

      - name: 🔍 Localizar y Organizar APK generado
        run: |
          mkdir -p output-apk
          APK_PATH=$(find android/app/build/outputs/apk/debug/ -name "*.apk" | head -n 1)
          if [ -f "$APK_PATH" ]; then
            echo "✅ APK generado exitosamente en: $APK_PATH"
            cp "$APK_PATH" output-apk/viajes-perdomo-debug.apk
            cp "$APK_PATH" output-apk/app-debug.apk
            
            # Preparar archivo ZIP con el APK
            cd output-apk
            zip -r ../viajes-perdomo-apk-bundle.zip .
            cd ..
            echo "✅ ZIP generado exitosamente: viajes-perdomo-apk-bundle.zip"
          else
            echo "❌ No se encontró el APK generado."
            exit 1
          fi

      - name: 📤 Subir APK individual a GitHub Artifacts
        uses: actions/upload-artifact@v4
        with:
          name: Viajes-Perdomo-APK
          path: output-apk/viajes-perdomo-debug.apk
          retention-days: 90

      - name: 📤 Subir archivo ZIP con APK
        uses: actions/upload-artifact@v4
        with:
          name: Viajes-Perdomo-APK-ZIP
          path: viajes-perdomo-apk-bundle.zip
          retention-days: 90
